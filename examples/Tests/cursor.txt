DECLARE EXTERNAL
INCLUDE "const.oph"

PROC main:
    REM text cursor
    PRINT "Hello";
    CURSOR ON
    GET

    REM graphics cursor
    gAT 50, 30
    gFONT KFontArialNormal32&
    gPRINT "Hello"
    gAT gX - 10, gY
    CURSOR gIDENTITY
    gUPDATE
    GET

    REM grey non flashing cursor
    gAT 50, 60
    gFONT KFontArialNormal32&
    gPRINT "Hello"
    gAT gX - 10, gY
    CURSOR gIDENTITY, 28, 10, 32, 6
    gUPDATE
    GET

    REM obloid cursor (sibo only)
    CURSOR gIDENTITY, 28, 10, 32, 1
    gUPDATE
    GET

ENDP
