PROC main:
    LOCAL ret%,sndHand%
    ret%=IOOPEN(sndHand%,"SND:",-1) REM open the device
    IF ret%<0
        PRINT "Failed to start"
        PRINT err$(err)
        GET
    ELSE
        icecream:(sndHand%)
        IOCLOSE(sndHand%)
    ENDIF
ENDP
